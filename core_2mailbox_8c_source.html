90void mailbox_free(struct Mailbox **ptr)
91{
92 if (!ptr || !*ptr)
93 return;
94
95 struct Mailbox *m = *ptr;
96
97 const bool do_free = (m->opened == 0) && !m->visible;
98
99 mutt_debug(LL_DEBUG3, "%sfreeing %s mailbox %s with refcount %d\n",
100 do_free ? "" : "not ", m->visible ? "visible" : "invisible",
101 buf_string(&m->pathbuf), m->opened);
102
103 if (!do_free)
104 {
105 return;
106 }
107
108 mutt_debug(LL_NOTIFY, "NT_MAILBOX_DELETE: %s %p\n",
109 mailbox_get_type_name(m->type), (void *) m);
110 struct EventMailbox ev_m = { m };
111 notify_send(m->notify, NT_MAILBOX, NT_MAILBOX_DELETE, &ev_m);
112
113 mutt_debug(LL_NOTIFY, "NT_EMAIL_DELETE_ALL\n");
114 struct EventEmail ev_e = { 0, NULL };
115 notify_send(m->notify, NT_EMAIL, NT_EMAIL_DELETE_ALL, &ev_e);
116
117 mailbox_clear_emails(m);
118 mutt_hash_free(&m->label_hash);
119 m->email_max = 0;
120
121 if (m->mdata_free && m->mdata)
122 m->mdata_free(&m->mdata);
123
124 buf_dealloc(&m->pathbuf);
125 cs_subset_free(&m->sub);
126 FREE(&m->name);
127 FREE(&m->realpath);
128 FREE(&m->emails);
129 FREE(&m->v2r);
130 notify_free(&m->notify);
131 mailbox_gc_run();
132
133 /* The NT_MAILBOX_DELETE notification might already have caused *ptr to be NULL,
134 * so call free() on the m pointer */
135 *ptr = NULL;
136 FREE(&m);
137}

175struct Mailbox *mailbox_find(const char *path)
176{
177 if (!path)
178 return NULL;
179
180 struct stat st = { 0 };
181 struct stat st_tmp = { 0 };
182
183 if (stat(path, &st) != 0)
184 return NULL;
185
186 struct MailboxArray ma = neomutt_mailboxes_get(NeoMutt, MUTT_MAILBOX_ANY);
187 struct Mailbox **mp = NULL;
188 struct Mailbox *m = NULL;
189 ARRAY_FOREACH(mp, &ma)
190 {
191 if ((stat(mailbox_path(*mp), &st_tmp) == 0) &&
192 (st.st_dev == st_tmp.st_dev) && (st.st_ino == st_tmp.st_ino))
193 {
194 m = *mp;
195 break;
196 }
197 }
198
199 ARRAY_FREE(&ma); // Clean up the ARRAY, but not the Mailboxes
200 return m;
201}

321void mailbox_gc_add(struct Email *e)
322{
323 ASSERT(e);
324 if (GC.idx == countof(GC.arr))
325 {
326 mailbox_gc_run();
327 }
328 GC.arr[GC.idx] = e;
329 GC.idx++;
330}
331
335void mailbox_gc_run(void)
336{
337 for (size_t i = 0; i < GC.idx; i++)
338 {
339 email_free(&GC.arr[i]);
340 }
341 GC.idx = 0;
342}
